import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 1. 데이터 파일 로드
# 파일이 스크립트와 같은 디렉토리에 있다고 가정합니다.
kospi_df = pd.read_csv('data/TimeSeries_3_KOSPI_Yearly_Monthly_Returns.csv')
kosdaq_df = pd.read_csv('data/TimeSeries_3_KOSDAQ_Yearly_Monthly_Returns.csv')

# 월별 컬럼 정의 (1월 ~ 12월)
months = [str(i) for i in range(1, 13)]

# 2. 각 지수별 월별 평균(Mean) 및 중앙값(Median) 계산
kospi_mean = kospi_df[months].mean()
kospi_median = kospi_df[months].median()

kosdaq_mean = kosdaq_df[months].mean()
kosdaq_median = kosdaq_df[months].median()


# 3. 개별 막대그래프 생성을 위한 범용 시각화 함수 정의
def draw_monthly_bar_chart(series, title, ylabel, color, save_filename):
    """
    series: 월별 통계 데이터 (Index가 '1'~'12')
    title: 그래프 제목
    ylabel: Y축 라벨
    color: 막대 색상 (HEX 코드 또는 색상명)
    save_filename: 저장할 파일 이름
    """
    # 가독성을 높이기 위해 수익률 크기순으로 내림차순 정렬
    sorted_series = series.sort_values(ascending=False)
    
    # 그래프 크기 설정
    fig, ax = plt.subplots(figsize=(10, 5))
    
    # X축 라벨 포맷 변경 ('1' -> 'Month 1')
    labels = [f"Month {m}" for m in sorted_series.index]
    x_positions = np.arange(len(labels))
    
    # 막대그래프 그리기
    bars = ax.bar(x_positions, sorted_series.values, color=color, alpha=0.8, edgecolor='black', linewidth=0.5)
    
    # 각 막대 위에 수치 라벨(%) 표시
    for bar in bars:
        height = bar.get_height()
        # 양수면 막대 위에, 음수면 막대 아래에 텍스트 위치 지정
        va = 'bottom' if height >= 0 else 'top'
        xytext = (0, 3) if height >= 0 else (0, -3)
        
        ax.annotate(f"{height:.2%}",
                    xy=(bar.get_x() + bar.get_width() / 2, height),
                    xytext=xytext,
                    textcoords="offset points",
                    ha='center', va=va, fontsize=9)
                    
    # 축 및 레이아웃 디테일 설정
    ax.set_xticks(x_positions)
    ax.set_xticklabels(labels, rotation=45, ha='right')
    ax.set_title(title, fontsize=13, fontweight='bold')
    ax.set_ylabel(ylabel, fontsize=11)
    
    # 0% 기준선 표시
    ax.axhline(0, color='black', linewidth=0.8, linestyle='-')
    ax.grid(True, linestyle='--', alpha=0.5, axis='y')
    
    # 라벨이 그래프 영역 밖으로 잘리지 않도록 Y축 여백 보정
    y_min, y_max = ax.get_ylim()
    ax.set_ylim(y_min * 1.2 if y_min < 0 else y_min, y_max * 1.2 if y_max > 0 else y_max)
    
    # 이미지 저장
    plt.tight_layout()
    plt.savefig(save_filename, dpi=150)
    plt.close()


# 4. 개별 그래프 4개 생성 및 파일 저장
# 코스피 (파란색 계열 지정)
draw_monthly_bar_chart(kospi_mean, 'KOSPI Monthly Return Mean (Sorted)', 'Mean Return', '#3232ff', 'kospi_monthly_mean.png')
draw_monthly_bar_chart(kospi_median, 'KOSPI Monthly Return Median (Sorted)', 'Median Return', '#3232ff', 'kospi_monthly_median.png')

# 코스닥 (빨간색 계열 지정)
draw_monthly_bar_chart(kosdaq_mean, 'KOSDAQ Monthly Return Mean (Sorted)', 'Mean Return', '#ff3232', 'kosdaq_monthly_mean.png')
draw_monthly_bar_chart(kosdaq_median, 'KOSDAQ Monthly Return Median (Sorted)', 'Median Return', '#ff3232', 'kosdaq_monthly_median.png')


# 5. [선택사항] 4개의 그래프를 하나의 이미지(2x2)로 모아서 비교하는 코드
fig, axs = plt.subplots(2, 2, figsize=(15, 10))

# (1) 코스피 평균
s_kospi_mean = kospi_mean.sort_values(ascending=False)
axs[0, 0].bar(np.arange(12), s_kospi_mean.values, color='#3232ff', alpha=0.8, edgecolor='black')
axs[0, 0].set_xticks(np.arange(12))
axs[0, 0].set_xticklabels([f"Month {m}" for m in s_kospi_mean.index], rotation=45, ha='right')
axs[0, 0].set_title('KOSPI Monthly Return Mean', fontweight='bold')
axs[0, 0].axhline(0, color='black', linewidth=0.8)
axs[0, 0].grid(True, linestyle='--', alpha=0.5)

# (2) 코스피 중앙값
s_kospi_median = kospi_median.sort_values(ascending=False)
axs[0, 1].bar(np.arange(12), s_kospi_median.values, color='#3232ff', alpha=0.8, edgecolor='black')
axs[0, 1].set_xticks(np.arange(12))
axs[0, 1].set_xticklabels([f"Month {m}" for m in s_kospi_median.index], rotation=45, ha='right')
axs[0, 1].set_title('KOSPI Monthly Return Median', fontweight='bold')
axs[0, 1].axhline(0, color='black', linewidth=0.8)
axs[0, 1].grid(True, linestyle='--', alpha=0.5)

# (3) 코스닥 평균
s_kosdaq_mean = kosdaq_mean.sort_values(ascending=False)
axs[1, 0].bar(np.arange(12), s_kosdaq_mean.values, color='#ff3232', alpha=0.8, edgecolor='black')
axs[1, 0].set_xticks(np.arange(12))
axs[1, 0].set_xticklabels([f"Month {m}" for m in s_kosdaq_mean.index], rotation=45, ha='right')
axs[1, 0].set_title('KOSDAQ Monthly Return Mean', fontweight='bold')
axs[1, 0].axhline(0, color='black', linewidth=0.8)
axs[1, 0].grid(True, linestyle='--', alpha=0.5)

# (4) 코스닥 중앙값
s_kosdaq_median = kosdaq_median.sort_values(ascending=False)
axs[1, 1].bar(np.arange(12), s_kosdaq_median.values, color='#ff3232', alpha=0.8, edgecolor='black')
axs[1, 1].set_xticks(np.arange(12))
axs[1, 1].set_xticklabels([f"Month {m}" for m in s_kosdaq_median.index], rotation=45, ha='right')
axs[1, 1].set_title('KOSDAQ Monthly Return Median', fontweight='bold')
axs[1, 1].axhline(0, color='black', linewidth=0.8)
axs[1, 1].grid(True, linestyle='--', alpha=0.5)

plt.tight_layout()
plt.savefig('combined_monthly_returns_analysis.png', dpi=150)
plt.show()
plt.close()

print("모든 막대그래프 시각화 파일 생성이 완료되었습니다.")
